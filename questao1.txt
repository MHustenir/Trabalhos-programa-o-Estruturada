#include <stdio.h>
#include <string.h>

#define NUM_ESTADOS 26

// Definição da estrutura para armazenar os dados de cada estado
typedef struct {
    char nome[50];
    int num_veiculos;
    int num_acidentes;
} Estado;

// (a) Procedimento para coletar os dados dos estados
void coletarDados(Estado cadastro[]) {
    printf("--- Coleta de Dados (%d Estados) ---\n", NUM_ESTADOS);
    for (int i = 0; i < NUM_ESTADOS; i++) {
        printf("\nEstado %d:\n", i + 1);
        printf("Nome do estado: ");
        // Limpar o buffer antes de ler a string
        setbuf(stdin, NULL);
        fgets(cadastro[i].nome, 50, stdin);
        // Remover a quebra de linha lida pelo fgets
        cadastro[i].nome[strcspn(cadastro[i].nome, "\n")] = '\0';
        
        printf("Numero de veiculos (em 2007): ");
        scanf("%d", &cadastro[i].num_veiculos);
        
        printf("Numero de acidentes (em 2007): ");
        scanf("%d", &cadastro[i].num_acidentes);
    }
}

// (b) Procedimento para encontrar os índices do maior e menor número de acidentes (retorno por referência)
void encontrarExtremos(Estado cadastro[], int *indMaior, int *indMenor) {
    *indMaior = 0;
    *indMenor = 0;
    
    for (int i = 1; i < NUM_ESTADOS; i++) {
        if (cadastro[i].num_acidentes > cadastro[*indMaior].num_acidentes) {
            *indMaior = i;
        }
        if (cadastro[i].num_acidentes < cadastro[*indMenor].num_acidentes) {
            *indMenor = i;
        }
    }
}

// (c) Função para calcular o percentual de veículos envolvidos em acidentes em um estado
float calcularPercentual(Estado cadastro[], int indiceEstado) {
    if (cadastro[indiceEstado].num_veiculos == 0) {
        return 0.0; // Evita divisão por zero
    }
    // Considera-se 1 acidente = 1 veículo envolvido para o cálculo deste percentual
    return ((float)cadastro[indiceEstado].num_acidentes / cadastro[indiceEstado].num_veiculos) * 100.0;
}

// (d) Função para calcular a média de acidentes no país
float calcularMediaPais(Estado cadastro[]) {
    int somaAcidentes = 0;
    for (int i = 0; i < NUM_ESTADOS; i++) {
        somaAcidentes += cadastro[i].num_acidentes;
    }
    return (float)somaAcidentes / NUM_ESTADOS;
}

// (e) Procedimento para exibir os estados com acidentes acima da média nacional
void exibirAcimaDaMedia(Estado cadastro[], float mediaPais) {
    printf("\n--- Estados com numero de acidentes acima da media nacional (%.2f) ---\n", mediaPais);
    int encontrou = 0;
    for (int i = 0; i < NUM_ESTADOS; i++) {
        if (cadastro[i].num_acidentes > mediaPais) {
            printf("- %s (%d acidentes)\n", cadastro[i].nome, cadastro[i].num_acidentes);
            encontrou = 1;
        }
    }
    if (!encontrou) {
        printf("Nenhum estado ficou acima da media.\n");
    }
}

int main() {
    Estado cadastroEstados[NUM_ESTADOS];
    int idxMaior, idxMenor;
    float mediaNacional;

    // a) Chama a coleta de dados
    coletarDados(cadastroEstados);

    // b) Encontra maior e menor número de acidentes e exibe no programa principal
    encontrarExtremos(cadastroEstados, &idxMaior, &idxMenor);
    printf("\n--- Resultados dos Extremos ---\n");
    printf("Maior numero de acidentes: %d (Estado: %s)\n", 
           cadastroEstados[idxMaior].num_acidentes, cadastroEstados[idxMaior].nome);
    printf("Menor numero de acidentes: %d (Estado: %s)\n", 
           cadastroEstados[idxMenor].num_acidentes, cadastroEstados[idxMenor].nome);

    // c) Exibe o percentual para cada estado
    printf("\n--- Percentual de Veiculos em Acidentes por Estado ---\n");
    for (int i = 0; i < NUM_ESTADOS; i++) {
        float percentual = calcularPercentual(cadastroEstados, i);
        printf("%s: %.2f%%\n", cadastroEstados[i].nome, percentual);
    }

    // d) Calcula a média e exibe no programa principal
    mediaNacional = calcularMediaPais(cadastroEstados);
    printf("\n--- Media Nacional ---\n");
    printf("A media de acidentes de transito no pais e: %.2f\n", mediaNacional);

    // e) Exibe estados acima da média
    exibirAcimaDaMedia(cadastroEstados, mediaNacional);

    return 0;
}
